      /* 29·ship c4 (G34, N11): a subscription is reused only when it was minted with the CURRENT signing key — one minted under an earlier key looks healthy and is refused by the push service on every message */
      return p3Vapid().then(function (key) {
        var verdict = existing ? rhKeyMatches(existing, key) : 'none';
        if (existing && verdict !== 'stale') { p3Log('sub_existing', { key: verdict }, 'ok'); return existing; }
        var drop = existing ? Promise.resolve(existing.unsubscribe()).then(function () { p3Log('sub_stale_dropped', {}, 'warn'); }, function () { p3Log('sub_stale_dropped', { failed: true }, 'warn'); }) : Promise.resolve();
        return drop.then(function () { return reg.pushManager.subscribe({ userVisibleOnly: true, applicationServerKey: p3B64ToBytes(key) }); });
      });
    });

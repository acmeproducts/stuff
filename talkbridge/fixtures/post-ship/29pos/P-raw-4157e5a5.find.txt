  if (!total) return 'You have ' + n + ' active room' + (n === 1 ? '' : 's') + ', there are no waiting messages.';
  var parts = [];
  if (chats) parts.push(chats + ' chat' + (chats === 1 ? '' : 's'));
  if (voice) parts.push(voice + ' voice call' + (voice === 1 ? '' : 's'));
  if (video) parts.push(video + ' video call' + (video === 1 ? '' : 's'));
  return 'Of ' + n + ' active room' + (n === 1 ? '' : 's') + ' there are ' + total +
         ' waiting message' + (total === 1 ? '' : 's') + ', ' + parts.join(' and ') + '.';
